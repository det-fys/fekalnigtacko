cmake_minimum_required(VERSION 3.22)

# Can also be built alone, without the renderer or game dependencies.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(CharacterCollisionTests LANGUAGES CXX)
    enable_testing()
    set(BUILD_BULLET2_DEMOS OFF CACHE BOOL "" FORCE)
    set(BUILD_CPU_DEMOS OFF CACHE BOOL "" FORCE)
    set(BUILD_OPENGL3_DEMOS OFF CACHE BOOL "" FORCE)
    set(BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
    set(BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
    set(BUILD_BULLET3 OFF CACHE BOOL "" FORCE)
    set(BUILD_PYBULLET OFF CACHE BOOL "" FORCE)
    set(INSTALL_LIBS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../external/bullet3" bullet)
endif()

add_executable(character_controller_tests character_controller_tests.cpp ../kinematic_character_controller.cpp)
target_compile_features(character_controller_tests PRIVATE cxx_std_17)
target_include_directories(character_controller_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../external/bullet3/src")
target_link_libraries(character_controller_tests PRIVATE BulletDynamics BulletCollision LinearMath)
if(USE_DOUBLE_PRECISION)
    target_compile_definitions(character_controller_tests PRIVATE BT_USE_DOUBLE_PRECISION)
endif()
add_test(NAME character_controller_tests COMMAND character_controller_tests)

# Optional comparison against the vendored implementation using identical scenes.
option(BUILD_BASELINE_COLLISION_TESTS "Build the original Bullet controller comparison" OFF)
if(BUILD_BASELINE_COLLISION_TESTS)
    add_executable(character_controller_baseline character_controller_tests.cpp)
    target_compile_features(character_controller_baseline PRIVATE cxx_std_17)
    target_include_directories(character_controller_baseline PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../external/bullet3/src")
    target_compile_definitions(character_controller_baseline PRIVATE TEST_ORIGINAL_CONTROLLER)
    target_link_libraries(character_controller_baseline PRIVATE BulletDynamics BulletCollision LinearMath)
    if(USE_DOUBLE_PRECISION)
        target_compile_definitions(character_controller_baseline PRIVATE BT_USE_DOUBLE_PRECISION)
    endif()
endif()
